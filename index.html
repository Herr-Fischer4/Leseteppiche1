function speechText(text) { const key = text.toLocaleLowerCase("de-DE"); if (key === "om") return "Ohm"; if (key === "me") return "Mee"; return text; } function getGermanVoice() { if (!("speechSynthesis" in window)) return null; const voices = window.speechSynthesis.getVoices(); return voices.find(v => v.lang.toLowerCase() === "de-de") || voices.find(v => v.lang.toLowerCase().startsWith("de-")) || voices.find(v => v.lang.toLowerCase() === "de") || null; } function speak(text, onEnd) { if (!("speechSynthesis" in window)) { if (onEnd) onEnd(); return; } window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(speechText(text)); utterance.lang = "de-DE"; utterance.rate = 0.78; const voice = getGermanVoice(); if (voice) utterance.voice = voice; if (onEnd) utterance.onend = onEnd; window.speechSynthesis.speak(utterance); }
